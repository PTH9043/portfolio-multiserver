cmake_minimum_required(VERSION 3.22)
project(GameProtocol LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(protobuf_MSVC_STATIC_RUNTIME OFF CACHE BOOL "" FORCE)
set(ABSL_ENABLE_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory(vendor/protobuf)
file(GLOB_RECURSE GENERATED_CPP CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Generated/Cpp/*.pb.cc")
if(GENERATED_CPP)
  add_library(GameProtocol STATIC ${GENERATED_CPP})
  target_include_directories(GameProtocol PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Generated/Cpp")
  target_link_libraries(GameProtocol PUBLIC protobuf::libprotobuf)
endif()
